- **localePrefix "always" (critical):** Every URL in the sitemap MUST come from `getPathname({ locale, href })`. Do not hardcode `/es/personas/...` or `/en/people/...`. The helper respects the routing pathnames config and stays in sync if keys change.
- **href keys only:** The `href` argument is the internal router key (e.g., `/people`, `/research`, `/people/[slug]`-style dynamic form `/people/${slug}`). NEVER pass the localized form (`/personas`, `/investigacion`). Pitfall 8 in RESEARCH.md.
- **sitemap lives at src/app/sitemap.ts** (NOT under `[locale]/`). It is a plain async function — no `setRequestLocale` call, no `getTranslations` call. Next.js 16 treats it as a file convention route.
- **robots lives at src/app/robots.ts** (root, not under `[locale]/`).
- **VERCEL_ENV (not NEXT_PUBLIC_VERCEL_ENV):** Use the server-only var to gate preview disallow. Pitfall in RESEARCH.md anti-patterns.
- **Person slugs included:** Enumerate the same subset that generateStaticParams in /people/[slug]/page.tsx produces — category pi | postdoc | phd (13 people per current roster). Non-clickable categories (undergrad, past) are NOT in the sitemap.

- **lastModified:** Use `new Date()` at build time — acceptable for a static export. Per-slug lastModified from content metadata is out of scope for Phase 5.
- **NAV-03:** Neither sitemap nor robots should emit any email address. (Trivially true; just don't add one.)
- **PERF-01:** Both files must remain SSG. Calling `getPeople()` and `siteConfig` is static module access — safe. Do NOT call `cookies()`, `headers()`, or any dynamic API.

    ```ts
    import type { MetadataRoute } from "next";
    import { getPathname } from "@/i18n/navigation";
    import { siteConfig } from "@/config/site";
    import { getPeople } from "@/content";

    const SITE_URL = siteConfig.url;

    // Static href keys are literal-typed so TypeScript narrows them to the concrete
    // pathnames config in routing.ts — no cast required.
    type StaticHref =
      | "/"
      | "/people"
      | "/research"
      | "/publications"
      | "/journal-club"
      | "/outreach"
      | "/contact";

    function staticEntry(href: StaticHref): MetadataRoute.Sitemap[number] {
      const esUrl = `${SITE_URL}${getPathname({ locale: "es", href })}`;
      const enUrl = `${SITE_URL}${getPathname({ locale: "en", href })}`;
      return {
        url: esUrl, // canonical default = Spanish (matches x-default)
        lastModified: new Date(),
        alternates: {
          languages: {
            es: esUrl,
            en: enUrl,
            "x-default": esUrl,
          },
        },
      };
    }

    // Dynamic `/people/[slug]` entries MUST use the object form of getPathname so
    // next-intl substitutes the slug against the pathnames map. The string-template
    // + cast pattern (`/people/${slug} as Href`) defeats the type system and creates
    // a trap if /people/[slug] ever gets renamed to wrap the slug segment in routing.ts.
    function personEntry(slug: string): MetadataRoute.Sitemap[number] {
      const esUrl = `${SITE_URL}${getPathname({
        locale: "es",
        href: { pathname: "/people/[slug]", params: { slug } },
      })}`;
      const enUrl = `${SITE_URL}${getPathname({
        locale: "en",
        href: { pathname: "/people/[slug]", params: { slug } },
      })}`;
      return {
        url: esUrl, // canonical default = Spanish (matches x-default)
        lastModified: new Date(),
        alternates: {
          languages: {
            es: esUrl,
            en: enUrl,
            "x-default": esUrl,
          },
        },
      };
    }

    export default function sitemap(): MetadataRoute.Sitemap {
      const staticHrefs: StaticHref[] = [
        "/",
        "/people",
        "/research",
        "/publications",
        "/journal-club",
        "/outreach",
        "/contact",
      ];

      const people = getPeople().filter(
        (p) => p.category === "pi" || p.category === "postdoc" || p.category === "phd"
      );

      return [
        ...staticHrefs.map(staticEntry),
        ...people.map((p) => personEntry(p.slug)),
      ];
    }
    ```
    Notes:
    - `getPeople` accessor is already exposed from `@/content` (verify via `rg 'export.*getPeople' src/content`). If the named export differs in the current content barrel, use the actual name; DO NOT introduce a new accessor.
    - Two entry helpers (static + person) split the job so each one uses the correct getPathname invocation form. The object form (`{ pathname: "/people/[slug]", params: { slug } }`) is the ONLY type-safe way to substitute slugs through next-intl — a plain string cast bypasses the pathnames map and silently emits the wrong URL if the dynamic route is ever renamed to wrap the slug (e.g. `/personas/perfil/[slug]`).

    ```ts
    import type { MetadataRoute } from "next";
    import { siteConfig } from "@/config/site";

    export default function robots(): MetadataRoute.Robots {
      // VERCEL_ENV is set by Vercel to "production" | "preview" | "development".
      // Use server-only var (NOT NEXT_PUBLIC_*) — keeps env data off the client.
      const isProduction = process.env.VERCEL_ENV === "production";

      return {
        rules: isProduction
          ? { userAgent: "*", allow: "/" }
          : { userAgent: "*", disallow: "/" },
        sitemap: `${siteConfig.url}/sitemap.xml`,
      };
    }
    ```
    Notes:
    - In local dev (`VERCEL_ENV` undefined) the result is `disallow: "/"`. That's fine because local dev is not crawled.
    - On Vercel preview deployments, VERCEL_ENV === "preview" → disallow prevents preview URLs from being indexed while the production URL is still being decided (CONTEXT.md open question).
    - On Vercel production, VERCEL_ENV === "production" → allow: "/" opens indexing.
  </action>
