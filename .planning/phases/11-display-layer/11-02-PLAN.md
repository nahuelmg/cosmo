    1. Create `src/components/publications/SourceFilter.tsx`:
       ```tsx
       'use client';

       import { useTranslations } from 'next-intl';

       export type SourceFilterValue = 'all' | 'inspirehep' | 'arxiv' | 'manual';

       interface SourceFilterProps {
         value: SourceFilterValue;
         onChange: (next: SourceFilterValue) => void;
       }

       const OPTIONS: { key: SourceFilterValue; i18nKey: string }[] = [
         { key: 'all', i18nKey: 'filter.all' },
         { key: 'inspirehep', i18nKey: 'filter.inspirehep' },
         { key: 'arxiv', i18nKey: 'filter.arxiv' },
         { key: 'manual', i18nKey: 'filter.manual' },
       ];

       export function SourceFilter({ value, onChange }: SourceFilterProps) {
         const t = useTranslations('publications');
         return (
           <div
             role="group"
             aria-label={t('filter.all')}
             className="mt-4 flex flex-wrap items-center gap-2"
           >
             {OPTIONS.map((opt) => {
               const active = value === opt.key;
               return (
                 <button
                   key={opt.key}
                   type="button"
                   aria-pressed={active}
                   onClick={() => {
                     // Non-Todos pills do not toggle off when clicked while active.
                     // Todos is the explicit all-state.
                     if (active && opt.key !== 'all') return;
                     onChange(opt.key);
                   }}
                   className={
                     active
                       ? 'rounded-full bg-accent px-3 py-1 text-sm font-medium text-white focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-accent-ring'
                       : 'rounded-full bg-surface-alt px-3 py-1 text-sm font-medium text-ink-muted hover:text-ink focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-accent-ring'
                   }
                 >
                   {t(opt.i18nKey)}
                 </button>
               );
             })}
           </div>
         );
       }
       ```
    2. The `aria-label` on the group uses `filter.all` ("Todos"/"All") as a stand-in; if a dedicated label key is desired later, it would need to be added in 11-01 — skip for now to preserve parallel safety.
  </instructions>

  <instructions>
    1. Create `src/components/publications/PublicationsClientShell.tsx`:
       ```tsx
       'use client';

       import { useMemo, useState } from 'react';
       import type { Publication } from '@/content';
       import { SourceFilter, type SourceFilterValue } from './SourceFilter';
       import { PublicationsYearGroup } from './PublicationsYearGroup';

       interface YearGroup {
         year: number;
         publications: Publication[];
       }

       interface PublicationsClientShellProps {
         groups: YearGroup[];
         memberSurnameSet: Set<string>;
         labels: {
           arxiv: string;
           doi: string;
           preprint: string;
           published: string;
         };
       }

       export function PublicationsClientShell({
         groups,
         memberSurnameSet,
         labels,
       }: PublicationsClientShellProps) {
         const [source, setSource] = useState<SourceFilterValue>('all');

         const filteredGroups = useMemo(() => {
           if (source === 'all') return groups;
           return groups
             .map((g) => ({
               year: g.year,
               publications: g.publications.filter((p) => p.source === source),
             }))
             .filter((g) => g.publications.length > 0);
         }, [groups, source]);

         return (
           <>
             <SourceFilter value={source} onChange={setSource} />
             {filteredGroups.map((g) => (
               <PublicationsYearGroup
                 key={g.year}
                 year={g.year}
                 publications={g.publications}
                 memberSurnameSet={memberSurnameSet}
                 labels={labels}
               />
             ))}
           </>
         );
       }
       ```
    2. `Set<string>` crosses the client boundary fine because this component is nested inside a server component that resolves at build time — the serialization pass still treats Sets as plain objects, but Next.js passes client component props through React Server Components protocol, which supports Set natively since 14.1. If build reports a non-serializable prop error, convert to `string[]` at the shell boundary and `new Set(arr)` inside; log this as a fallback but prefer the Set form.
    3. No memoization of `buildMemberSurnameSet` is needed here — constructed once at page render, passed in by the server.
  </instructions>

    `pnpm tsc --noEmit` clean for this file. Component compiles and is importable.
