```tsx
import Link from 'next/link';
import {getTranslations} from 'next-intl/server';
import {siteConfig} from '@/config/site';
import {EmailLink} from '@/components/ui/EmailLink';

interface SiteFooterProps {
  locale: string;
}

export async function SiteFooter({locale}: SiteFooterProps) {
  const tFooter = await getTranslations({locale, namespace: 'footer'});

  // Split the canonical contact email into the two props <EmailLink> expects.
  // siteConfig.contactEmail is validated to always contain '@' by 02-01's
  // TypeScript shape; splitOnce is safe.
  const [user, domain] = siteConfig.contactEmail.split('@');

  const currentYear = new Date().getFullYear();

  return (
    <footer
      className={[
        'mt-auto',                // pushes footer to bottom in flex-col body
        'bg-surface-alt',
        'shadow-sm',              // subtle separator from main content (no border)
      ].join(' ')}
    >
      <div
        className={[
          'mx-auto max-w-5xl',
          'px-6 py-10',
          'grid gap-8',
          'md:grid-cols-2',
        ].join(' ')}
      >
        {/* Identity column */}
        <div className="flex flex-col gap-3">
          <h2 className="font-serif text-lg font-semibold text-ink">
            {siteConfig.groupName}
          </h2>
          <ul className="flex flex-col gap-1 text-sm text-ink-muted">
            <li>{tFooter('affiliationUba')}</li>
            <li>{tFooter('affiliationFcen')}</li>
            <li>{tFooter('affiliationConicet')}</li>
          </ul>
        </div>

        {/* Contact column */}
        <div className="flex flex-col gap-3 md:items-end">
          <EmailLink
            user={user}
            domain={domain}
            className={[
              'text-sm text-ink-muted hover:text-accent transition-colors',
              'focus-visible:outline-none',
              'focus-visible:ring-2 focus-visible:ring-accent-ring rounded',
            ].join(' ')}
          />

          {siteConfig.socialLinks.length > 0 && (
            <ul className="flex flex-wrap gap-4 text-sm text-ink-muted md:justify-end">
              {siteConfig.socialLinks.map((link) => (
                <li key={link.url}>
                  <Link
                    href={link.url}
                    target="_blank"
                    rel="noopener noreferrer"
                    aria-label={link.label}
                    className={[
                      'hover:text-accent transition-colors',
                      'focus-visible:outline-none',
                      'focus-visible:ring-2 focus-visible:ring-accent-ring rounded',
                    ].join(' ')}
                  >
                    {link.label}
                  </Link>
                </li>
              ))}
            </ul>
          )}
        </div>
      </div>

      <div className="mx-auto max-w-5xl px-6 pb-6">
        <p className="text-xs text-ink-subtle">
          {tFooter('copyright', {year: currentYear})}
        </p>
      </div>
    </footer>
  );
}
```

Layout reasoning:
- Two-column grid at `md+`, single column stacked below (CONTEXT.md — Claude's discretion; two-column is the most common academic pattern and reads clean at the site's 5xl max-width).
- `mt-auto` + the layout's flex column (set in Task 2) pin the footer to the bottom on short pages.
- Affiliations as a `<ul>` of translated strings from the `footer` namespace (already exists — 01-03 seeded these keys). The research open-question about using `siteConfig.affiliations` is answered here: we use the translation keys instead because they're already bilingual and the siteConfig affiliations duplicate that. `affiliations` in siteConfig stays available for Phase 4's footer-less institutional references and for potential linked-logos in v2.
- Social section is ALWAYS guarded with `.length > 0` — currently `siteConfig.socialLinks = []` so the section renders nothing (RESEARCH.md Codebase State note). No fallback text, no "follow us" heading — just nothing, which is the academic-restraint choice.
- Copyright uses the already-localized `footer.copyright` key with a `{year}` placeholder — next-intl supports ICU-style interpolation natively; no extra work needed.
- No UBA/FCEN/CONICET logos (CONTEXT.md deferred decision — text only in v1).
- No postal address (CONTEXT.md — lives on Contact page, do not duplicate).

```tsx
import type {ReactNode} from 'react';
import {NextIntlClientProvider, hasLocale} from 'next-intl';
import {setRequestLocale} from 'next-intl/server';
import {notFound} from 'next/navigation';
import {routing} from '@/i18n/routing';
import {fontSerif, fontSans} from '@/app/fonts';
import {SkipLink} from '@/components/layout/SkipLink';
import {SiteHeader} from '@/components/layout/SiteHeader';
import {SiteFooter} from '@/components/layout/SiteFooter';
import '@/app/globals.css';

export function generateStaticParams() {
  return routing.locales.map((locale) => ({locale}));
}

type Props = {
  children: ReactNode;
  params: Promise<{locale: string}>;
};

export default async function LocaleLayout({children, params}: Props) {
  const {locale} = await params;

  if (!hasLocale(routing.locales, locale)) {
    notFound();
  }

  setRequestLocale(locale);

  return (
    <html lang={locale} className={`${fontSerif.variable} ${fontSans.variable}`}>
      <body className="bg-surface text-ink antialiased min-h-screen flex flex-col">
        <NextIntlClientProvider>
          <SkipLink locale={locale} />
          <SiteHeader />
          <main id="main-content" tabIndex={-1} className="flex-1">
            {children}
          </main>
          <SiteFooter locale={locale} />
        </NextIntlClientProvider>
      </body>
    </html>
  );
}
```

Key changes from Plan 01's version:
- Body gains `min-h-screen flex flex-col` so the footer's `mt-auto` pushes it to the bottom.
- `<main>` gains `flex-1` so it consumes vertical space, pushing the footer below the fold on short pages.
- `<SiteHeader />` sits between SkipLink and main (so Tab order is: SkipLink → Header nav → main content → Footer links).
- `<SiteFooter locale={locale} />` receives the locale explicitly since it's a server component calling `getTranslations({locale, namespace: 'footer'})`.

Preserve everything else (translations, `setRequestLocale`, Greek probe if present).

Locate the `<main` opening and `</main>` closing tags in `page.tsx` and rename both to `<section` / `</section>` (a section with a heading is a valid landmark-free content container). Do NOT delete the content inside — Phase 4 will rebuild it properly; this plan only stops the landmark collision.
